stage('BDD2OCTANE') {
    steps.withCredentials([[$class: 'UsernamePasswordMultiBinding', credentialsId: env.SQUAD_NEXUS3_CREDENTIAL_ID, usernameVariable: 'USERNAME', passwordVariable: 'PASSWORD']]) {
        steps.sh """
            if [ ! -d target/entity-results ] || [ -z "\$(ls -A target/entity-results/TEST-*.xml 2>/dev/null)" ]; then
                echo "ERROR: target/entity-results has no TEST-*.xml files. Skipping entity push."
            else
                mvn bdd2octane:run -DreportFiles=\\"**/target/entity-results/TEST-*.xml\\" -DfeatureFiles=\\"**/src/test/resources/features/\\" -Dframework=\\"cucumber-jvm\\" -DresultFile=\\"entity-cucumber-jvm-result.xml\\" -Drepo.login=${env.USERNAME} -Drepo.pwd=${env.PASSWORD}
            fi
        """

        steps.sh """
            if [ ! -d target/facade-results ] || [ -z "\$(ls -A target/facade-results/TEST-*.xml 2>/dev/null)" ]; then
                echo "ERROR: target/facade-results has no TEST-*.xml files. Skipping facade push."
            else
                mvn bdd2octane:run -DreportFiles=\\"**/target/facade-results/TEST-*.xml\\" -DfeatureFiles=\\"**/src/test/resources/features/\\" -Dframework=\\"cucumber-jvm\\" -DresultFile=\\"facade-cucumber-jvm-result.xml\\" -Drepo.login=${env.USERNAME} -Drepo.pwd=${env.PASSWORD}
            fi
        """
    }
}

stage('publishGherkinResults') {
    publishGherkinResults '**/entity-cucumber-jvm-result.xml, **/facade-cucumber-jvm-result.xml'
}
